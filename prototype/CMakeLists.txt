cmake_minimum_required(VERSION 3.24)
project(sbmdr_c0 LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
if(DEFINED VCPKG_TARGET_TRIPLET)
    find_package(boost_beast CONFIG REQUIRED)
    set(SBMDR_BOOST Boost::beast)
else()
    find_package(Boost 1.83 CONFIG REQUIRED)
    set(SBMDR_BOOST Boost::headers)
endif()
find_package(Threads REQUIRED)
find_package(absl CONFIG REQUIRED)
pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET libcurl simdjson nlohmann_json openssl sqlite3 libzstd)
add_executable(sbmdr-c0 main.cpp)
target_compile_features(sbmdr-c0 PRIVATE cxx_std_20)
option(SBMDR_COUNT_ALLOCATIONS "Count C++ new calls for allocation diagnostics" OFF)
if(NOT SBMDR_COUNT_ALLOCATIONS)
    target_compile_definitions(sbmdr-c0 PRIVATE SBMDR_NO_NEW_COUNTER)
endif()
target_compile_options(sbmdr-c0 PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(sbmdr-c0 PRIVATE ${SBMDR_BOOST} Threads::Threads absl::crc32c PkgConfig::DEPS)
option(SBMDR_DIAGNOSTICS "Enable bounded per-stage timing diagnostics" OFF)
if(SBMDR_DIAGNOSTICS)
    target_compile_definitions(sbmdr-c0 PRIVATE SBMDR_DIAGNOSTICS)
endif()
enable_testing()
set(PYTHON_BASELINE "" CACHE PATH "Checkout of the fixed Python Recorder commit")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME c0-loopback COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify.py"
    "$<TARGET_FILE:sbmdr-c0>" "${PYTHON_BASELINE}")
add_test(NAME c1-recorder COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_recorder.py"
    "$<TARGET_FILE:sbmdr-c0>" "${PYTHON_BASELINE}")
add_test(NAME c2-reliability COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_reliability.py"
    "$<TARGET_FILE:sbmdr-c0>" "${PYTHON_BASELINE}")
add_test(NAME c2-handoff-limit COMMAND "$<TARGET_FILE:sbmdr-c0>" --check-handoff)
add_test(NAME c2-core COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_core.py"
    "$<TARGET_FILE:sbmdr-c0>" "${PYTHON_BASELINE}")
add_test(NAME c2-depth-edges COMMAND "$<TARGET_FILE:sbmdr-c0>" --check-depth-bridge)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT CMAKE_CXX_FLAGS MATCHES "fsanitize")
    # LD_PRELOAD syscall interposition runs in the unsanitized Release process.
    # The real recorder/queue/bridge suites run separately with ASan/UBSan/TSan.
    add_library(sbmdr-fault-io SHARED fault_io.cpp)
    target_link_libraries(sbmdr-fault-io PRIVATE ${CMAKE_DL_LIBS})
    add_test(NAME c2-recovery-boundaries COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_SOURCE_DIR}/verify_recovery_boundaries.py"
        "$<TARGET_FILE:sbmdr-c0>" "${PYTHON_BASELINE}" "$<TARGET_FILE:sbmdr-fault-io>")
endif()
add_test(NAME c3-configured COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_configured.py"
    "$<TARGET_FILE:sbmdr-c0>" "${PYTHON_BASELINE}")
add_test(NAME c4-candles COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify_candles.py")
set_tests_properties(c4-candles PROPERTIES ENVIRONMENT "PYTHONPATH=${PYTHON_BASELINE}/src")
