cmake_minimum_required(VERSION 3.24)
project(sbmdr_c0 LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
find_package(Boost 1.83 CONFIG REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET libcurl simdjson nlohmann_json absl_crc32c openssl)
add_executable(sbmdr-c0 main.cpp)
target_compile_features(sbmdr-c0 PRIVATE cxx_std_20)
target_compile_options(sbmdr-c0 PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(sbmdr-c0 PRIVATE Boost::headers Threads::Threads PkgConfig::DEPS)
enable_testing()
set(PYTHON_BASELINE "" CACHE PATH "Checkout of the fixed Python Recorder commit")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME c0-loopback COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/verify.py"
    "$<TARGET_FILE:sbmdr-c0>" "${PYTHON_BASELINE}")
