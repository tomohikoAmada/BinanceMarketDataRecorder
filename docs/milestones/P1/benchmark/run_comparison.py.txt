import json
import os
import subprocess
from pathlib import Path

root = Path('/Users/amada/Documents/Development/Crypto/BinanceMarketDataRecorder')
evidence = Path('/var/tmp/bmdr-p1-20261027-8JCF7QMA')
python = root / '.venv/bin/python'
versions = {'A': (evidence / 'before', '6f8fe1e802d1e2b99e4f0a38a1f5cacfe707079a'),
            'B': (root, 'ce1826b')}
for number, side in enumerate('ABBA', 1):
    repo, revision = versions[side]
    name = f'all-features-{number}-{side}'
    env = dict(os.environ, PYTHONPATH=f'{repo / "src"}:{repo}')
    command = [str(python), str(repo / 'docs/milestones/P0/steady_baseline.py'),
               '--repo', str(repo), '--commit', revision, '--symbols', '2',
               '--warmup', '20', '--seconds', '120', '--output', str(evidence / f'{name}.json')]
    with (evidence / f'{name}.log').open('x') as log:
        print(f'START {name}', flush=True)
        outcome = subprocess.run(command, env=env, stdout=log, stderr=subprocess.STDOUT)
    print(f'END {name} exit={outcome.returncode}', flush=True)
    if outcome.returncode:
        raise SystemExit(outcome.returncode)
